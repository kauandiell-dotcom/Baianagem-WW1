# Event 1: The Zimmermann Telegram
# -------------------------------------------------------------
country_event = {
	id = ww1_germany_naval.1
	title = ww1_germany_naval.1.t
	desc = ww1_germany_naval.1.d
	picture = GFX_report_event_german_speech

	is_triggered_only = yes
	fire_only_once = yes

	option = {
		name = ww1_germany_naval.1.a # Send the telegram proposing alliance to Mexico (MEX)
		ai_chance = { factor = 90 }
		set_country_flag = GER_zimmermann_telegram_dispatched
		if = {
			limit = { country_exists = MEX }
			MEX = {
				country_event = { id = ww1_germany_naval.2 days = 14 }
			}
		}
	}

	option = {
		name = ww1_germany_naval.1.b # Do not risk American intervention
		ai_chance = { factor = 10 }
		add_political_power = -20
	}
}

# -------------------------------------------------------------
# Event 50: American Ultimatum - Unrestricted Submarine Warfare Crisis
# -------------------------------------------------------------
country_event = {
	id = ww1_germany_naval.50
	title = ww1_germany_naval.50.t
	desc = ww1_germany_naval.50.d
	picture = GFX_report_event_american_doughboys_parade

	is_triggered_only = yes
	fire_only_once = yes

	# Option A: A State of War Exists! (Historical)
	option = {
		name = ww1_germany_naval.50.a
		ai_chance = { factor = 95 }
		add_war_support = 0.35
		add_stability = 0.10
		if = {
			limit = { country_exists = GER NOT = { has_war_with = GER } }
			declare_war_on = {
				target = GER
				type = annex_everything
			}
		}
		if = {
			limit = { country_exists = ENG ENG = { is_faction_leader = yes } }
			ENG = { add_to_faction = ROOT }
		}
		if = {
			limit = { country_exists = FRA FRA = { is_faction_leader = yes } NOT = { is_in_faction_with = FRA } }
			FRA = { add_to_faction = ROOT }
		}
		news_event = { id = ww1_germany_naval.51 days = 1 }
	}

	# Option B: Demand German Guarantees and Issue Stern Protest
	option = {
		name = ww1_germany_naval.50.b
		ai_chance = { factor = 5 }
		add_political_power = 50
		add_war_support = -0.15
		if = {
			limit = { country_exists = GER }
			GER = {
				add_to_variable = { GER_us_intervention_tension = -30 }
				ger_clamp_naval_gauges = yes
			}
		}
	}
}
