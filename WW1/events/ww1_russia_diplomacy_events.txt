# -------------------------------------------------------------
# Event 1: German Deliberation on Soviet Peace Feelers
# Triggered by Soviet Decree on Peace (SOV_decree_on_peace)
# -------------------------------------------------------------
country_event = {
	id = ww1_russia_diplomacy.1
	title = ww1_russia_diplomacy.1.t
	desc = ww1_russia_diplomacy.1.d
	picture = GFX_event_WW1_ger_russian_opportunity

	is_triggered_only = yes

	# Option A: Demand Harsh Brest-Litovsk Terms
	option = {
		name = ww1_russia_diplomacy.1.a
		ai_chance = { factor = 95 }
		SOV = {
			country_event = { id = ww1_russia_diplomacy.2 days = 2 }
		}
		custom_effect_tooltip = GER_dictate_brest_litovsk_tt
	}

	# Option B: Reject Overtures & Push for Military Victory
	option = {
		name = ww1_russia_diplomacy.1.b
		ai_chance = { factor = 5 }
		add_war_support = 0.05
		custom_effect_tooltip = GER_refuse_peace_talks_tt
	}
}

# -------------------------------------------------------------
# Event 2: Soviet Decision on German Brest-Litovsk Terms
# -------------------------------------------------------------
country_event = {
	id = ww1_russia_diplomacy.2
	title = ww1_russia_diplomacy.2.t
	desc = ww1_russia_diplomacy.2.d
	picture = GFX_event_WW1_sov_sovnarkom

	is_triggered_only = yes

	# Choice A: Sign the Treaty of Brest-Litovsk
	option = {
		name = ww1_russia_diplomacy.2.a
		ai_chance = { factor = 85 }

		# White peace with Central Powers
		white_peace = GER
		if = { limit = { country_exists = AUS } white_peace = AUS }
		if = { limit = { country_exists = TUR } white_peace = TUR }
		if = { limit = { country_exists = BUL } white_peace = BUL }

		set_global_flag = SOV_brest_litovsk_signed
		GER = {
			add_ideas = GER_brest_litovsk_deliveries
		}

		add_stability = 0.05
		add_political_power = 50
		custom_effect_tooltip = SOV_brest_litovsk_signed_tt
	}

	# Choice B: "Neither war nor peace" - Trotsky walks out
	option = {
		name = ww1_russia_diplomacy.2.b
		ai_chance = { factor = 15 }

		set_country_flag = SOV_trotsky_neither_war_nor_peace
		custom_effect_tooltip = SOV_trotsky_walkout_tt

		GER = {
			country_event = { id = ww1_russia_diplomacy.3 days = 2 }
		}
	}
}

# Event 20: 1918 Armistice - Dissolution of the Treaty of Brest-Litovsk!
# -------------------------------------------------------------
news_event = {
	id = ww1_russia_diplomacy.20
	title = ww1_russia_diplomacy.20.t
	desc = ww1_russia_diplomacy.20.d
	picture = GFX_event_WW1_ger_republic

	major = yes
	is_triggered_only = yes
	fire_only_once = yes

	immediate = {
		SOV_dissolve_brest_litovsk = yes
	}

	# Option A: Soviet Russia perspective
	option = {
		name = ww1_russia_diplomacy.20.a
		trigger = { tag = SOV }
		add_political_power = 100
		add_war_support = 0.10
		custom_effect_tooltip = SOV_target_vistula_reclamation_tt
	}

	# Option B: German perspective
	option = {
		name = ww1_russia_diplomacy.20.b
		trigger = { tag = GER }
		add_stability = -0.05
	}

	# Option C: World perspective
	option = {
		name = ww1_russia_diplomacy.20.c
		trigger = {
			NOT = {
				OR = {
					tag = SOV
					tag = GER
				}
			}
		}
	}
}
