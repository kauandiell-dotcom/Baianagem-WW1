add_namespace = ww1_germany_endsieg

# =========================================================================
# ww1_germany_endsieg.1: The Fall of the Rhine & The Versailles Crisis
# =========================================================================
country_event = {
	id = ww1_germany_endsieg.1
	title = ww1_germany_endsieg.1.t
	desc = ww1_germany_endsieg.1.d
	picture = GFX_report_event_german_troops

	fire_only_once = yes

	trigger = {
		tag = GER
		has_war = yes
		NOT = { has_country_flag = GER_endsieg_activated }
		NOT = { has_country_flag = GER_versailles_signed }
		NOT = { has_government = communism }
		OR = {
			# Catastrophic military catastrophe: Berlin occupied and deep collapse
			AND = {
				surrender_progress > 0.70
				any_enemy_country = {
					controls_state = 64 # Berlin / Brandenburg
				}
			}
			# Late-war historical collapse (Late 1918) with Western Front breach or revolution
			AND = {
				date > 1918.8.1
				surrender_progress > 0.35
				OR = {
					any_enemy_country = {
						OR = {
							controls_state = 42 # Rhineland
							controls_state = 51 # Moselland
							controls_state = 55 # Nassau / Frankfurt
							controls_state = 64 # Berlin / Brandenburg
						}
					}
					has_country_flag = GER_republic_proclaimed
					has_country_flag = GER_revolutionary_unrest
					check_variable = { GER_discontent > 75 }
				}
			}
			# November 1918 Revolution capitulation request
			AND = {
				date > 1918.10.15
				has_country_flag = GER_republic_proclaimed
			}
			# Severe mid-to-late war total exhaustion collapse
			AND = {
				date > 1917.6.1
				surrender_progress > 0.75
			}
		}
	}

	mean_time_to_happen = {
		days = 5
	}

	# Option A: Historical Versailles Armistice - Proclaim the Weimar Republic
	option = {
		name = ww1_germany_endsieg.1.a
		ai_chance = {
			factor = 70
		}
		set_country_flag = GER_versailles_signed
		set_politics = {
			ruling_party = democratic
			elections_allowed = yes
		}
		add_stability = -0.20
		add_war_support = -0.40
		every_enemy_country = {
			white_peace = GER
		}
		# Territorial cessions mandated by the Treaty of Versailles
		if = {
			limit = { country_exists = FRA }
			FRA = { transfer_state = 28 } # Alsace-Lorraine
		}
		if = {
			limit = { country_exists = BEL }
			BEL = { transfer_state = 1085 } # Eupen-Malmedy
		}
		if = {
			limit = { NOT = { country_exists = POL } }
			release = POL
		}
		if = {
			limit = { country_exists = POL }
			POL = {
				transfer_state = 86 # Posen
				transfer_state = 85 # Danzig / West Prussia corridor
			}
		}
		add_ideas = GER_versailles_reparations
	}

	# Option B: Defy the World! - Total Resistance (Endsieg / Götterdämmerung)
	option = {
		name = ww1_germany_endsieg.1.b
		ai_chance = {
			factor = 30
		}
		set_country_flag = GER_endsieg_activated
		load_focus_tree = ww1_germany_endsieg_focus
		load_oob = "GER_volkssturm_endsieg"
		add_war_support = 0.25
		add_stability = -0.10
		add_timed_idea = {
			idea = GER_endsieg_fanaticism
			days = 730
		}
	}
}
