# ww1_diplomacy.1: Austria-Hungary receives German Blank Cheque
country_event = {
	id = ww1_diplomacy.1
	title = ww1_diplomacy.1.t
	desc = ww1_diplomacy.1.d
	picture = GFX_report_event_german_speech

	is_triggered_only = yes

	# Option A: Deliver the July Ultimatum to Serbia (Historical)
	option = {
		name = ww1_diplomacy.1.a
		ai_chance = { base = 95 }
		add_war_support = 0.15
		add_political_power = 50
		set_country_flag = aus_blank_cheque_accepted
		if = {
			limit = { country_exists = SER }
			SER = {
				country_event = { id = ww1_diplomacy.2 days = 2 }
			}
		}
		if = {
			limit = { country_exists = GER }
			GER = {
				country_event = { id = ww1_diplomacy.3 days = 1 }
			}
		}
	}

	# Option B: Seek diplomatic de-escalation (Alt-History)
	option = {
		name = ww1_diplomacy.1.b
		ai_chance = { base = 5 }
		add_stability = 0.10
		add_war_support = -0.10
		if = {
			limit = { country_exists = GER }
			GER = {
				country_event = { id = ww1_diplomacy.4 days = 1 }
			}
		}
	}
}
