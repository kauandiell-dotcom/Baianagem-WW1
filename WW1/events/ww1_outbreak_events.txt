add_namespace = ww1_outbreak

# Political responses belong to the July Crisis. Event suffixes must be numeric.
news_event = {
	id = ww1_outbreak.1
	title = ww1_outbreak.1.t
	desc = ww1_outbreak.1.d
	picture = GFX_event_ww1_sarajevo_assassination
	major = yes
	fire_only_once = yes
	trigger = {
		tag = AUS
		date > 1914.6.27
		NOT = { has_global_flag = sarajevo_assassination_occurred }
		country_exists = SER
	}
	mean_time_to_happen = { days = 1 }
	immediate = {
		set_global_flag = sarajevo_assassination_occurred
		AUS = {
			if = {
				limit = { NOT = { has_country_flag = ww1_july_crisis_started } }
				set_country_flag = ww1_july_crisis_started
				country_event = { id = ww1_diplomacy.1 days = 3 }
			}
		}
	}
	option = {
		name = ww1_outbreak.1.a
		trigger = { tag = AUS }
	}
	option = {
		name = ww1_outbreak.1.b
		trigger = { tag = SER }
	}
	option = {
		name = ww1_outbreak.1.c
		trigger = { NOT = { OR = { tag = AUS tag = SER } } }
	}
}
