	}

# -------------------------------------------------------------
# Event 10: Red Victory (Bolshevik Triumph)
# -------------------------------------------------------------
country_event = {
	id = ww1_russia_civil_war.10
	title = ww1_russia_civil_war.10.t
	desc = ww1_russia_civil_war.10.d
	picture = GFX_event_WW1_sov_civil_war

	trigger = {
		tag = SOV
		has_country_flag = SOV_russian_civil_war_active
		country_exists = WHT
		has_war_with = WHT
		OR = {
			WHT = { surrender_progress > 0.85 }
			AND = {
				controls_state = 571 # Omsk
				controls_state = 398 # Ekaterinburg
				controls_state = 217 # Tsaritsyn
			}
		}
	}

	mean_time_to_happen = { days = 3 }
	fire_only_once = yes

	option = {
		name = ww1_russia_civil_war.10.a
		ai_chance = { factor = 100 }
		annex_country = { target = WHT transfer_troops = yes }
		clr_country_flag = SOV_russian_civil_war_active
		set_rule = { can_join_factions = yes }
		add_stability = 0.20
		add_political_power = 150
	}
}
