# -------------------------------------------------------------------------
# ww1_agadir.1: Panthersprung — SMS Panther Arrives at Agadir
# -------------------------------------------------------------------------
country_event = {
	id = ww1_agadir.1
	title = ww1_agadir.1.t
	desc = ww1_agadir.1.d
	picture = GFX_report_event_german_speech

	fire_only_once = yes
	is_triggered_only = yes

	trigger = {
		tag = GER
		NOT = { has_global_flag = agadir_crisis_happened }
	}

	immediate = {
		set_global_flag = agadir_crisis_happened
		set_global_flag = agadir_crisis_active
		clr_global_flag = ww1_agadir_dispatch_pending
		set_variable = { global.ww1_agadir_crisis_tension = 25 }
	}

	option = {
		name = ww1_agadir.1.a
		ai_chance = { factor = 90 }
		add_political_power = 50
		if = {
			limit = { country_exists = FRA }
			FRA = { country_event = { id = ww1_agadir.2 days = 2 } }
		}
		if = {
			limit = { country_exists = ENG }
			ENG = { country_event = { id = ww1_agadir.3 days = 3 } }
		}
	}

	option = {
		name = ww1_agadir.1.b
		ai_chance = { factor = 10 }
		add_stability = 0.05
		country_event = { id = ww1_agadir.30 days = 1 }
	}
}
