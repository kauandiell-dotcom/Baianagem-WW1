# =========================================================================
# ww1_austria_diplomacy.1: The Colonel Redl Espionage Affair (May 1913)
# =========================================================================
country_event = {
	id = ww1_austria_diplomacy.1
	title = ww1_austria_diplomacy.1.t
	desc = ww1_austria_diplomacy.1.d
	picture = GFX_event_AUH_ww1_council

	fire_only_once = yes

	trigger = {
		tag = AUS
		date > 1913.5.1
		date < 1913.12.1
		NOT = { has_war = yes }
		NOT = { has_country_flag = AUH_colonel_redl_affair_fired }
	}

	mean_time_to_happen = {
		days = 10
	}

	# Option A: Overhaul mobilization Plan B/R (costs 50 PP, removes Russian intel advantage)
	option = {
		name = ww1_austria_diplomacy.1.a
		set_country_flag = AUH_colonel_redl_affair_fired
		set_country_flag = AUH_redl_plans_overhauled
		add_political_power = -50
		add_stability = 0.05
		custom_effect_tooltip = ww1_austria_diplomacy.1.a_tt
		ai_will_do = {
			factor = 80
		}
	}

	# Option B: Cover up suicide to preserve Evidenzbureau honor (-10% army defense vs Russia in 1914)
	option = {
		name = ww1_austria_diplomacy.1.b
		set_country_flag = AUH_colonel_redl_affair_fired
		set_country_flag = AUH_redl_evidenzbureau_coverup
		add_political_power = 25
		add_stability = -0.05
		custom_effect_tooltip = ww1_austria_diplomacy.1.b_tt
		ai_will_do = {
			factor = 20
		}
	}
}

# =========================================================================
# ww1_austria_diplomacy.20: The Austro-Polish Solution
# =========================================================================
country_event = {
	id = ww1_austria_diplomacy.20
	title = ww1_austria_diplomacy.20.t
	desc = ww1_austria_diplomacy.20.d
	picture = GFX_event_AUH_ww1_council

	fire_only_once = yes

	trigger = {
		tag = AUS
		OR = {
			controls_state = 10
			has_global_flag = poland_liberated
		}
		NOT = { has_country_flag = AUH_austro_polish_solution_fired }
	}

	mean_time_to_happen = {
		days = 15
	}

	# Option A: Crown Archduke Karl Stephan as King of Poland (creates Polish client state)
	option = {
		name = ww1_austria_diplomacy.20.a
		set_country_flag = AUH_austro_polish_solution_fired
		set_country_flag = AUH_austro_polish_solution_proclaimed
		add_political_power = 50
		add_stability = 0.05
		if = {
			limit = {
				NOT = { country_exists = POL }
			}
			POL = {
				transfer_state = 10
				transfer_state = 98
				transfer_state = 90
				transfer_state = 92
				leave_faction = yes
				prevent_join_factions = yes
				set_rule = { can_join_factions = no }
				set_rule = { can_create_factions = no }
				set_cosmetic_tag = POL_AUS_danubian_state
			}
			puppet = POL
		}
		else = {
			if = {
				limit = {
					country_exists = POL
					NOT = { POL = { is_subject_of = AUS } }
				}
				puppet = POL
				POL = {
					set_cosmetic_tag = POL_AUS_danubian_state
					set_rule = { can_join_factions = no }
					set_rule = { can_create_factions = no }
				}
			}
		}
		custom_effect_tooltip = ww1_austria_diplomacy.20.a_tt
		ai_will_do = {
			factor = 70
		}
	}

	# Option B: Yield administration to German General Government of Warsaw
	option = {
		name = ww1_austria_diplomacy.20.b
		set_country_flag = AUH_austro_polish_solution_fired
		set_country_flag = AUH_german_polish_solution_yielded
		add_political_power = 25
		add_opinion_modifier = { target = GER modifier = AUH_ww1_alliance_renewed }
		custom_effect_tooltip = ww1_austria_diplomacy.20.b_tt
		ai_will_do = {
			factor = 30
		}
	}
}

# =========================================================================
# ww1_austria_diplomacy.30: The Brotfrieden (Bread Peace, February 1918)
# =========================================================================
country_event = {
	id = ww1_austria_diplomacy.30
	title = ww1_austria_diplomacy.30.t
	desc = ww1_austria_diplomacy.30.d
	picture = GFX_event_AUH_ww1_council

	fire_only_once = yes

	trigger = {
		tag = AUS
		has_war = yes
		date > 1918.1.1
		check_variable = { AUH_vienna_grain < 40 }
		NOT = { has_country_flag = AUH_brotfrieden_fired }
	}

	mean_time_to_happen = {
		days = 10
	}

	# Option A: Conclude agricultural deliveries from Ukraine (adds 40 AUH_vienna_grain, clamps variables)
	option = {
		name = ww1_austria_diplomacy.30.a
		set_country_flag = AUH_brotfrieden_fired
		set_country_flag = AUH_brotfrieden_concluded
		add_to_variable = { AUH_vienna_grain = 40 }
		add_to_variable = { AUH_war_weariness = -10 }
		add_stability = 0.05
		AUH_clamp_dashboard_variables = yes
		custom_effect_tooltip = ww1_austria_diplomacy.30.a_tt

			factor = 90
		}
	}

	# Option B: Refuse diplomatic concessions
	option = {
		name = ww1_austria_diplomacy.30.b
		set_country_flag = AUH_brotfrieden_fired
		set_country_flag = AUH_brotfrieden_refused
		add_stability = -0.10
		add_to_variable = { AUH_war_weariness = 15 }
		AUH_clamp_dashboard_variables = yes
