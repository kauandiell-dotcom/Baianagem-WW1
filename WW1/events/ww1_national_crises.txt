# ALEMANHA: Inverno dos Nabos & O Bloqueio da Entente
# -------------------------------------------------------------
country_event = {
	id = ww1_crisis.1
	title = ww1_crisis.1.t
	desc = ww1_crisis.1.d
	picture = GFX_report_event_generic_strike

	is_triggered_only = yes
	fire_only_once = yes

	trigger = {
		tag = GER
		date > 1914.8.1
		has_war = yes
		OR = {
			has_war_with = FRA
			has_war_with = ENG
		}
	}

	immediate = {
		set_country_flag = GER_crisis_event_fired
	}

	option = {
		name = ww1_crisis.1.a
		add_ideas = GER_turnip_winter_crisis_3
		set_country_flag = GER_crisis_active
		custom_effect_tooltip = ww1_crisis_ger_tooltip
	}
}
