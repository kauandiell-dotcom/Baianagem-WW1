# Consecutive material pressure, recovery and episode cleanup. COUNTRY scope.

ww1_GER_close_crisis_episode = {
 remove_ideas = GER_turnip_winter_crisis_1
 remove_ideas = GER_turnip_winter_crisis_2
 remove_ideas = GER_turnip_winter_crisis_3
 clr_country_flag = GER_crisis_active
 clr_country_flag = GER_crisis_triggered
 clr_country_flag = GER_crisis_resolved
 clr_country_flag = GER_relief_step_1_done
 clr_country_flag = GER_relief_step_2_done
 clr_country_flag = GER_relief_step_3_done
 set_country_flag = { flag = GER_crisis_recovery_grace days = 180 }
 set_variable = { GER_pressure_weeks = 0 }
 set_variable = { GER_recovery_weeks = 0 }
}

ww1_GER_update_crisis = {
 if = {
  limit = { NOT = { has_country_flag = ww1_society_initialized } }
  set_country_flag = ww1_society_initialized
  set_variable = { GER_war_weeks = 0 }
  set_variable = { GER_pressure_weeks = 0 }
  set_variable = { GER_recovery_weeks = 0 }
 }
 # Native country dynamic variables: equipment actually present/required in deployed armies.
 set_variable = { ww1_required_infantry_equipment = num_target_equipment_in_armies@infantry_equipment }
 set_variable = { ww1_infantry_equipment_ratio = 1 }
 if = {
  limit = { check_variable = { ww1_required_infantry_equipment > 0 } }
  set_variable = { ww1_infantry_equipment_ratio = num_equipment_in_armies@infantry_equipment }
  divide_variable = { ww1_infantry_equipment_ratio = ww1_required_infantry_equipment }
  clamp_variable = { var = ww1_infantry_equipment_ratio min = 0 max = 1 }
 }
 if = {
  limit = { has_war = yes any_enemy_country = { is_major = yes } }
  add_to_variable = { GER_war_weeks = 1 }
  if = {
   limit = { ww1_ger_crisis_pressure = yes }
   add_to_variable = { GER_pressure_weeks = 1 }
   set_variable = { GER_recovery_weeks = 0 }
   if = {
    limit = {
     check_variable = { GER_war_weeks > 25 }
     check_variable = { GER_pressure_weeks > 3 }
     NOT = { has_country_flag = GER_crisis_active }
     NOT = { has_country_flag = GER_crisis_recovery_grace }
    }
    clr_country_flag = GER_relief_step_1_done
    clr_country_flag = GER_relief_step_2_done
    clr_country_flag = GER_relief_step_3_done
    set_variable = { GER_pressure_weeks = 0 }
    country_event = { id = ww1_crisis.1 }
   }
   if = {
    limit = { has_country_flag = GER_crisis_active check_variable = { GER_pressure_weeks > 23 } NOT = { has_idea = GER_turnip_winter_crisis_3 } }
    remove_ideas = GER_turnip_winter_crisis_1
    remove_ideas = GER_turnip_winter_crisis_2
    add_ideas = GER_turnip_winter_crisis_3
   }
   else_if = {
    limit = { has_country_flag = GER_crisis_active check_variable = { GER_pressure_weeks > 11 } has_idea = GER_turnip_winter_crisis_1 }
    remove_ideas = GER_turnip_winter_crisis_1
    add_ideas = GER_turnip_winter_crisis_2
   }
  }
  else = {
   set_variable = { GER_pressure_weeks = 0 }
   add_to_variable = { GER_recovery_weeks = 1 }
   if = {
    limit = { has_country_flag = GER_crisis_active check_variable = { GER_recovery_weeks > 5 } }
    set_variable = { GER_recovery_weeks = 0 }
    if = { limit = { has_idea = GER_turnip_winter_crisis_3 } swap_ideas = { remove_idea = GER_turnip_winter_crisis_3 add_idea = GER_turnip_winter_crisis_2 } }
    else_if = { limit = { has_idea = GER_turnip_winter_crisis_2 } swap_ideas = { remove_idea = GER_turnip_winter_crisis_2 add_idea = GER_turnip_winter_crisis_1 } }
    else = { ww1_GER_close_crisis_episode = yes }
   }
  }
 }
 else = {
  if = { limit = { has_country_flag = GER_crisis_active } ww1_GER_close_crisis_episode = yes set_country_flag = ww1_reconstruction_available }
  set_variable = { GER_war_weeks = 0 }
  set_variable = { GER_pressure_weeks = 0 }
  set_variable = { GER_recovery_weeks = 0 }
 }
}
