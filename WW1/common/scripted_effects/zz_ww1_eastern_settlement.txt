# COUNTRY scope GER, once weekly. No stale terms survive peace, disappearance or a changed interlocutor.
ww1_GER_update_eastern_negotiations = {
 if = {
  limit = {
   has_country_flag = ww1_eastern_terms_authorised
   NOT = {
    OR = {
     AND = { has_country_flag = ww1_eastern_terms_for_SOV country_exists = SOV has_war_with = SOV }
     AND = { has_country_flag = ww1_eastern_terms_for_RUS country_exists = RUS has_war_with = RUS }
    }
   }
  }
  clr_country_flag = ww1_eastern_terms_authorised
  clr_country_flag = ww1_eastern_terms_for_SOV
  clr_country_flag = ww1_eastern_terms_for_RUS
 }
 if = { limit = { country_exists = SOV NOT = { has_war_with = SOV } } SOV = { clr_country_flag = ww1_eastern_negotiations_started } }
 if = { limit = { country_exists = RUS NOT = { has_war_with = RUS } } RUS = { clr_country_flag = ww1_eastern_negotiations_started } }
}
