 set_variable = { ww1_britain_irish_tension = 25 }
 set_variable = { ww1_britain_labour_support = 15 }
 set_variable = { ww1_britain_irish_exemption_factor = 0 }
 set_variable = { ww1_britain_ulster_tension = 10 }
 set_variable = { ww1_britain_dominion_consent = 50 }
 remove_ideas = ENG_professional_bef
 add_ideas = ENG_ww1_professional_service
 recruit_character = ENG_ww1_david_lloyd_george
 recruit_character = ENG_ww1_andrew_bonar_law
 recruit_character = ENG_ww1_ramsay_macdonald
}

ww1_britain_recalculate_debt = {
 clamp_variable = { var = ww1_britain_debt_burden min = 0 max = 100 }
 if = {
  limit = { check_variable = { ww1_britain_debt_burden > 69 } }
  remove_ideas = ENG_ww1_debt_pressure_1
  remove_ideas = ENG_ww1_debt_pressure_2
  if = { limit = { NOT = { has_idea = ENG_ww1_debt_pressure_3 } } add_ideas = ENG_ww1_debt_pressure_3 }
 }
 else_if = {
  limit = { check_variable = { ww1_britain_debt_burden > 39 } }
  remove_ideas = ENG_ww1_debt_pressure_1
  remove_ideas = ENG_ww1_debt_pressure_3
  if = { limit = { NOT = { has_idea = ENG_ww1_debt_pressure_2 } } add_ideas = ENG_ww1_debt_pressure_2 }
 }
 else_if = {
  limit = { check_variable = { ww1_britain_debt_burden > 19 } }
  remove_ideas = ENG_ww1_debt_pressure_2
  remove_ideas = ENG_ww1_debt_pressure_3
  if = { limit = { NOT = { has_idea = ENG_ww1_debt_pressure_1 } } add_ideas = ENG_ww1_debt_pressure_1 }
 }
 else = {
  remove_ideas = ENG_ww1_debt_pressure_1
  remove_ideas = ENG_ww1_debt_pressure_2
  remove_ideas = ENG_ww1_debt_pressure_3
 }
}

ww1_britain_update_irish_exemption = {
 # Preserve voluntary service in Ireland when Great Britain adopts compulsory service.
 set_variable = { ww1_britain_actual_recruitment = modifier@conscription }
 set_variable = { ww1_britain_irish_exemption_factor = 0 }
 if = {
  limit = {
   has_country_flag = ENG_ww1_military_service_authorised
   NOT = { has_country_flag = ENG_ww1_irish_conscription_authorised }
   check_variable = { ww1_britain_actual_recruitment > 0.015 }
  }
  set_variable = { ww1_britain_irish_exemption_factor = .015 }
  divide_variable = { ww1_britain_irish_exemption_factor = ww1_britain_actual_recruitment }
  subtract_from_variable = { ww1_britain_irish_exemption_factor = 1 }
  clamp_variable = { var = ww1_britain_irish_exemption_factor min = -.95 max = 0 }
  113 = { if = { limit = { is_owned_by = ENG } add_dynamic_modifier = { modifier = ww1_britain_irish_exemption scope = ENG } } }
  134 = { if = { limit = { is_owned_by = ENG } add_dynamic_modifier = { modifier = ww1_britain_irish_exemption scope = ENG } } }
  135 = { if = { limit = { is_owned_by = ENG } add_dynamic_modifier = { modifier = ww1_britain_irish_exemption scope = ENG } } }
  119 = { if = { limit = { is_owned_by = ENG } add_dynamic_modifier = { modifier = ww1_britain_irish_exemption scope = ENG } } }
 }
 else = {
  113 = { remove_dynamic_modifier = ww1_britain_irish_exemption }
  134 = { remove_dynamic_modifier = ww1_britain_irish_exemption }
  135 = { remove_dynamic_modifier = ww1_britain_irish_exemption }
  119 = { remove_dynamic_modifier = ww1_britain_irish_exemption }
 }
}

ww1_britain_weekly_administration = {
 if = {
  limit = { has_war = yes }
  set_country_flag = ENG_ww1_war_participated
  if = { limit = { has_country_flag = ENG_ww1_home_rule_deferred } add_to_variable = { ww1_britain_irish_tension = .35 } }
 }
 else = {
  remove_ideas = ENG_ww1_war_cabinet
  remove_ideas = ENG_ww1_munitions_ministry
  remove_ideas = ENG_ww1_emergency_controls
  remove_ideas = ENG_ww1_kitchener_armies
  remove_ideas = ENG_ww1_pals_battalions
  remove_ideas = ENG_ww1_flanders_supply
  remove_ideas = ENG_ww1_creeping_barrage
  remove_ideas = ENG_ww1_convoy_system
  remove_ideas = ENG_ww1_munitions_contracts
  remove_ideas = ENG_ww1_shell_inspection
  remove_ideas = ENG_ww1_labour_dilution
  remove_ideas = ENG_ww1_food_control
  remove_ideas = ENG_ww1_coal_allocation
  if = { limit = { has_country_flag = ENG_ww1_home_rule_devolved } add_to_variable = { ww1_britain_irish_tension = -.15 } }
 }
 if = {
  limit = {
   OR = { has_idea = limited_conscription has_idea = extensive_conscription has_idea = service_by_requirement has_idea = all_adults_serve has_idea = scraping_the_barrel }
   NOT = { has_country_flag = ENG_ww1_military_service_authorised }
  }
  if = { limit = { NOT = { has_idea = ENG_ww1_unsanctioned_conscription } } add_ideas = ENG_ww1_unsanctioned_conscription }
 }
 else = { remove_ideas = ENG_ww1_unsanctioned_conscription }
 if = {
  limit = { has_country_flag = ENG_ww1_home_rule_devolved }
  add_to_variable = { ww1_britain_irish_tension = -.15 }
 }
 clamp_variable = { var = ww1_britain_irish_tension min = 0 max = 100 }
 clamp_variable = { var = ww1_britain_labour_support min = 0 max = 100 }
 clamp_variable = { var = ww1_britain_ulster_tension min = 0 max = 100 }
 clamp_variable = { var = ww1_britain_dominion_consent min = 0 max = 100 }
 if = { limit = { check_variable = { ww1_britain_ulster_tension > 69 } } add_to_variable = { ww1_britain_irish_tension = .2 } }
 if = {
  limit = {
   check_variable = { ww1_britain_irish_tension > 69 }
   NOT = { has_country_flag = ENG_ww1_irish_crisis_cooldown }
   113 = { is_owned_by = ENG }
  }
  set_country_flag = { flag = ENG_ww1_irish_crisis_cooldown days = 180 }
  country_event = { id = ww1_britain.3 }
 }
 ww1_britain_recalculate_debt = yes
 ww1_britain_update_irish_exemption = yes
}
