# Scripted Effects for Dynamic Independent Multi-Front Troop Counter (Baianagem-WW1)

bww1_fc_cleanup_frontlines = {
	every_state = {
		limit = {
			has_variable = bww1_is_frontline_state
		}
		clear_variable = bww1_is_frontline_state
		clear_variable = bww1_state_friendly_manpower
		clear_variable = bww1_state_enemy_manpower
		clear_variable = bww1_state_friendly_divs
		clear_variable = bww1_state_enemy_divs
	}
	clear_array = ROOT.bww1_frontline_friendly_states
	clear_array = ROOT.bww1_frontline_enemy_states
}

bww1_fc_scan_and_update_frontlines = {
	# 1. Clear previous frontline markings
	bww1_fc_cleanup_frontlines = yes

	# 2. Calculate average manpower per division for ROOT
	set_temp_variable = { friendly_mp_per_div = ROOT.deployed_army_manpower_k }
	multiply_temp_variable = { friendly_mp_per_div = 1000 }
	if = {
		limit = {
			check_variable = { ROOT.num_divisions > 0 }
		}
		divide_temp_variable = { friendly_mp_per_div = ROOT.num_divisions }
	}
	else = {
		set_temp_variable = { friendly_mp_per_div = 16000 }
	}
	clamp_temp_variable = { var = friendly_mp_per_div min = 5000 max = 22000 }

	# 3. Scan ALL controlled states of ROOT to detect every active border sector across all frontiers
	every_controlled_state = {
		limit = {
			any_neighbor_state = {
				NOT = { controller = { tag = ROOT } }
				NOT = { controller = { is_in_faction_with = ROOT } }
			}
		}

		# Tag friendly state on the active frontline
		set_variable = { bww1_is_frontline_state = 1 }
		ROOT = {
			add_to_array = { bww1_frontline_friendly_states = PREV }
		}

		# Count friendly divisions in this state
		set_variable = { bww1_state_friendly_divs = 0 }
		set_temp_variable = { t_divs = ROOT.num_armies_in_state@THIS }
		if = {
			limit = {
				check_variable = { t_divs > 0 }
			}
			set_variable = { bww1_state_friendly_divs = t_divs }
		}
		else = {
			if = { limit = { controller = { divisions_in_state = { state = PREV size > 40 } } } set_variable = { bww1_state_friendly_divs = 42 } }
			else_if = { limit = { controller = { divisions_in_state = { state = PREV size > 30 } } } set_variable = { bww1_state_friendly_divs = 32 } }
			else_if = { limit = { controller = { divisions_in_state = { state = PREV size > 20 } } } set_variable = { bww1_state_friendly_divs = 22 } }
			else_if = { limit = { controller = { divisions_in_state = { state = PREV size > 15 } } } set_variable = { bww1_state_friendly_divs = 16 } }
			else_if = { limit = { controller = { divisions_in_state = { state = PREV size > 10 } } } set_variable = { bww1_state_friendly_divs = 11 } }
			else_if = { limit = { controller = { divisions_in_state = { state = PREV size > 7 } } } set_variable = { bww1_state_friendly_divs = 8 } }
			else_if = { limit = { controller = { divisions_in_state = { state = PREV size > 4 } } } set_variable = { bww1_state_friendly_divs = 5 } }
			else_if = { limit = { controller = { divisions_in_state = { state = PREV size > 2 } } } set_variable = { bww1_state_friendly_divs = 3 } }
			else_if = { limit = { controller = { divisions_in_state = { state = PREV size > 0 } } } set_variable = { bww1_state_friendly_divs = 1 } }
		}

		# Calculate real friendly manpower stationed in this state
		set_variable = { bww1_state_friendly_manpower = bww1_state_friendly_divs }
		multiply_variable = { bww1_state_friendly_manpower = friendly_mp_per_div }

		# 4. Scan opposing neighbor states directly facing this sector
		every_neighbor_state = {
			limit = {
				NOT = { controller = { tag = ROOT } }
				NOT = { controller = { is_in_faction_with = ROOT } }
				NOT = { has_variable = bww1_is_frontline_state }
			}

			set_variable = { bww1_is_frontline_state = 1 }
			ROOT = {
				add_to_array = { bww1_frontline_enemy_states = PREV }
			}

			# Calculate foreign controller's average manpower per division
			set_temp_variable = { enemy_mp_per_div = controller.deployed_army_manpower_k }
			multiply_temp_variable = { enemy_mp_per_div = 1000 }
			if = {
				limit = {
					check_variable = { controller.num_divisions > 0 }
				}
				divide_temp_variable = { enemy_mp_per_div = controller.num_divisions }
			}
			else = {
				set_temp_variable = { enemy_mp_per_div = 14000 }
			}
			clamp_temp_variable = { var = enemy_mp_per_div min = 5000 max = 22000 }

			# Count enemy divisions in this foreign state
			set_variable = { bww1_state_enemy_divs = 0 }
			set_temp_variable = { t_edivs = controller.num_armies_in_state@THIS }
			if = {
				limit = {
					check_variable = { t_edivs > 0 }
				}
				set_variable = { bww1_state_enemy_divs = t_edivs }
			}
			else = {
				if = { limit = { controller = { divisions_in_state = { state = PREV size > 40 } } } set_variable = { bww1_state_enemy_divs = 42 } }
				else_if = { limit = { controller = { divisions_in_state = { state = PREV size > 30 } } } set_variable = { bww1_state_enemy_divs = 32 } }
				else_if = { limit = { controller = { divisions_in_state = { state = PREV size > 20 } } } set_variable = { bww1_state_enemy_divs = 22 } }
				else_if = { limit = { controller = { divisions_in_state = { state = PREV size > 15 } } } set_variable = { bww1_state_enemy_divs = 16 } }
				else_if = { limit = { controller = { divisions_in_state = { state = PREV size > 10 } } } set_variable = { bww1_state_enemy_divs = 11 } }
				else_if = { limit = { controller = { divisions_in_state = { state = PREV size > 7 } } } set_variable = { bww1_state_enemy_divs = 8 } }
				else_if = { limit = { controller = { divisions_in_state = { state = PREV size > 4 } } } set_variable = { bww1_state_enemy_divs = 5 } }
				else_if = { limit = { controller = { divisions_in_state = { state = PREV size > 2 } } } set_variable = { bww1_state_enemy_divs = 3 } }
				else_if = { limit = { controller = { divisions_in_state = { state = PREV size > 0 } } } set_variable = { bww1_state_enemy_divs = 1 } }
			}

			# Calculate real enemy manpower stationed in this state
			set_variable = { bww1_state_enemy_manpower = bww1_state_enemy_divs }
			multiply_variable = { bww1_state_enemy_manpower = enemy_mp_per_div }
		}
	}
}
