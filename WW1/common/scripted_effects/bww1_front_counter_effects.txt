# Scripted Effects for Frontline Troop Counter (Baianagem-WW1)

bww1_fc_cleanup_frontlines = {
	every_state = {
		limit = {
			has_variable = bww1_is_frontline_state
		}
		clear_variable = bww1_is_frontline_state
		clear_variable = bww1_friendly_manpower
		clear_variable = bww1_enemy_manpower
		clear_variable = bww1_friendly_divs
		clear_variable = bww1_enemy_divs
		clear_variable = bww1_intel_level
	}
}

bww1_fc_scan_and_update_frontlines = {
	# Clear previously tagged frontline states
	bww1_fc_cleanup_frontlines = yes

	if = {
		limit = {
			has_war = yes
		}

		# Scan friendly states controlled directly by player
		every_controlled_state = {
			limit = {
				any_neighbor_state = {
					controller = {
						has_war_with = ROOT
					}
				}
			}
			set_variable = { bww1_is_frontline_state = 1 }
			bww1_fc_calculate_state_manpower = yes
		}

		# Scan states controlled by active coalition allies participating in this war
		every_allied_country = {
			limit = {
				has_war_together_with = ROOT
			}
			every_controlled_state = {
				limit = {
					any_neighbor_state = {
						controller = {
							has_war_with = ROOT
						}
					}
				}
				set_variable = { bww1_is_frontline_state = 1 }
				bww1_fc_calculate_state_manpower = yes
			}
		}
	}
}

bww1_fc_calculate_state_manpower = {
	# 1. Initialize friendly division count
	set_variable = { bww1_friendly_divs = 0 }

	if = {
		limit = {
			controller = {
				divisions_in_state = {
					state = PREV
					size > 40
				}
			}
		}
		set_variable = { bww1_friendly_divs = 48 }
	}
	else_if = {
		limit = {
			controller = {
				divisions_in_state = {
					state = PREV
					size > 30
				}
			}
		}
		set_variable = { bww1_friendly_divs = 36 }
	}
	else_if = {
		limit = {
			controller = {
				divisions_in_state = {
					state = PREV
					size > 20
				}
			}
		}
		set_variable = { bww1_friendly_divs = 25 }
	}
	else_if = {
		limit = {
			controller = {
				divisions_in_state = {
					state = PREV
					size > 14
				}
			}
		}
		set_variable = { bww1_friendly_divs = 18 }
	}
	else_if = {
		limit = {
			controller = {
				divisions_in_state = {
					state = PREV
					size > 9
				}
			}
		}
		set_variable = { bww1_friendly_divs = 12 }
	}
	else_if = {
		limit = {
			controller = {
				divisions_in_state = {
					state = PREV
					size > 5
				}
			}
		}
		set_variable = { bww1_friendly_divs = 8 }
	}
	else_if = {
		limit = {
			controller = {
				divisions_in_state = {
					state = PREV
					size > 2
				}
			}
		}
		set_variable = { bww1_friendly_divs = 4 }
	}
	else_if = {
		limit = {
			controller = {
				divisions_in_state = {
					state = PREV
					size > 0
				}
			}
		}
		set_variable = { bww1_friendly_divs = 2 }
	}
	else = {
		set_variable = { bww1_friendly_divs = 1 }
	}

	# Check co-belligerent allies in this state
	ROOT = {
		every_allied_country = {
			limit = {
				has_war_together_with = ROOT
			}
			if = {
				limit = {
					divisions_in_state = {
						state = PREV
						size > 10
					}
				}
				PREV = { add_to_variable = { bww1_friendly_divs = 12 } }
			}
			else_if = {
				limit = {
					divisions_in_state = {
						state = PREV
						size > 3
					}
				}
				PREV = { add_to_variable = { bww1_friendly_divs = 5 } }
			}
			else_if = {
				limit = {
					divisions_in_state = {
						state = PREV
						size > 0
					}
				}
				PREV = { add_to_variable = { bww1_friendly_divs = 2 } }
			}
		}
	}

	# Convert friendly divisions to manpower (average 16,850 soldiers per WW1 division)
	set_variable = { bww1_friendly_manpower = bww1_friendly_divs }
	multiply_variable = { bww1_friendly_manpower = 16850 }
	# Add organic variance based on state ID so numbers look like real field reports (e.g. 684,392)
	set_temp_variable = { t_friendly_var = state_id }
	multiply_temp_variable = { t_friendly_var = 137 }
	clamp_temp_variable = { var = t_friendly_var min = 240 max = 1980 }
	add_to_variable = { bww1_friendly_manpower = t_friendly_var }

	# 2. Opposing enemy forces across the frontline
	set_variable = { bww1_enemy_divs = 0 }

	every_neighbor_state = {
		limit = {
			controller = {
				has_war_with = ROOT
			}
		}
		set_temp_variable = { t_neighbor_divs = 1 }
		if = {
			limit = {
				controller = {
					divisions_in_state = {
						state = PREV
						size > 30
					}
				}
			}
			set_temp_variable = { t_neighbor_divs = 36 }
		}
		else_if = {
			limit = {
				controller = {
					divisions_in_state = {
						state = PREV
						size > 20
					}
				}
			}
			set_temp_variable = { t_neighbor_divs = 24 }
		}
		else_if = {
			limit = {
				controller = {
					divisions_in_state = {
						state = PREV
						size > 10
					}
				}
			}
			set_temp_variable = { t_neighbor_divs = 14 }
		}
		else_if = {
			limit = {
				controller = {
					divisions_in_state = {
						state = PREV
						size > 4
					}
				}
			}
			set_temp_variable = { t_neighbor_divs = 7 }
		}
		else_if = {
			limit = {
				controller = {
					divisions_in_state = {
						state = PREV
						size > 0
					}
				}
			}
			set_temp_variable = { t_neighbor_divs = 3 }
		}

		PREV = {
			add_to_variable = { bww1_enemy_divs = t_neighbor_divs }
		}
	}

	# Ensure enemy divisions minimum is 1 for frontline states
	if = {
		limit = {
			check_variable = { bww1_enemy_divs < 1 }
		}
		set_variable = { bww1_enemy_divs = 2 }
	}

	# 3. Calculate estimated enemy manpower
	set_variable = { bww1_enemy_manpower = bww1_enemy_divs }
	multiply_variable = { bww1_enemy_manpower = 16850 }
	set_temp_variable = { t_enemy_var = state_id }
	multiply_temp_variable = { t_enemy_var = 251 }
	clamp_temp_variable = { var = t_enemy_var min = 315 max = 2470 }
	add_to_variable = { bww1_enemy_manpower = t_enemy_var }
}
