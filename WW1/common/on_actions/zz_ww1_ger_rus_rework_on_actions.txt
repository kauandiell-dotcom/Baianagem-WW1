on_actions = {
	# Master Weekly Simulation Hook (Country-Scoped by Engine)
	on_weekly = {
		effect = {
			# Soviet / Russian Crisis & Revolutionary Gauges
			if = {
				limit = { original_tag = SOV }
				ww1_sov_rework_weekly = yes
				if = {
					limit = {
						NOT = { has_variable = SOV_revolutionary_tension }
					}
					sov_russia_init_crisis_gauges = yes
				}
				sov_russia_weekly_crisis_update = yes
			}

			# German Empire Wartime Operations & Trench Stalemate
			if = {
				limit = { original_tag = GER }
				ww1_ger_rework_weekly = yes
				ger_uboat_weekly_tension_update = yes

				# Western Front Trench Stalemate Transition Hook
				if = {
					limit = {
						has_war_with = FRA
						has_country_flag = ger_schlieffen_plan_launched
						NOT = { controls_state = 16 }
						NOT = { has_idea = GER_schlieffen_momentum }
						NOT = { has_idea = GER_western_trench_stalemate }
					}
					country_event = { id = ww1_germany.10 }
				}

				# 1918 German Armistice Dissolution Hook
				if = {
					limit = {
						date > 1918.11.1
						has_global_flag = SOV_brest_litovsk_signed
						GER = { has_capitulated = yes }
					}
					SOV_dissolve_brest_litovsk = yes
					news_event = { id = ww1_russia_diplomacy.20 }
				}
			}

			# Austria-Hungary Crown Dashboard & Grain Simulation Hook
			if = {
				limit = { tag = AUS }
				if = {
					limit = {
						NOT = { has_variable = AUH_vienna_grain }
					}
					AUH_init_crown_dashboard = yes
				}
				AUH_clamp_dashboard_variables = yes
				AUH_weekly_grain_tick = yes
			}

			# French Republic Dashboard & Wartime Survival Vectors Hook
			if = {
				limit = { original_tag = FRA }
				if = {
					limit = {
						NOT = { has_country_flag = fra_dashboard_initialized }
					}
					FRA_init_republic_dashboard = yes
					set_country_flag = fra_dashboard_initialized
				}
				FRA_clamp_dashboard_variables = yes
				FRA_weekly_republic_tick = yes
			}

			# Kingdom of Serbia Balance of Power & Variables Clamping Hook
			if = {
				limit = { original_tag = SER }
				if = {
					limit = {
						NOT = { has_variable = ser_civil_military_balance }
					}
					SER_init_power_balance = yes
				}
				SER_weekly_bop_tick = yes

				# Great Retreat & Salonika Sanctuary Automation Hook
				if = {
					limit = {
						has_war = yes
						NOT = { controls_state = 107 }
						NOT = { has_country_flag = ser_executed_great_retreat }
					}
					country_event = { id = ww1_serbia_retreat.1 }
				}
			}
		}
	}

	# 1918 German Armistice Dissolution Hook
	on_surrender = {
		effect = {
			if = {
				limit = {
					tag = GER
					has_global_flag = SOV_brest_litovsk_signed
				}
				SOV_dissolve_brest_litovsk = yes
				news_event = { id = ww1_russia_diplomacy.20 }
			}
		}
	}

	# Transient Memory & Garbage Collection Hook
	on_monthly = {
		effect = {
			ww1_clean_transient_memory = yes
		}
	}
}
