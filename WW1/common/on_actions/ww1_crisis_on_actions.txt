# Country-specific weekly callbacks: each country advances exactly once per game week.
# The generic on_weekly hook executes in each country's scope, so it must not sweep tags.
on_actions = {
 on_weekly_GER = {
  effect = {
   ww1_GER_update_crisis = yes
   ww1_GER_update_eastern_negotiations = yes
   if = {
    limit = { date > 1911.7.1 NOT = { has_global_flag = agadir_crisis_happened } NOT = { has_global_flag = ww1_agadir_dispatch_pending } }
    set_global_flag = ww1_agadir_dispatch_pending
    country_event = { id = ww1_agadir.1 days = 1 }
   }
  }
 }
 on_weekly_ENG = {
  effect = {
   ww1_ENG_update_crisis = yes
  }
 }
 on_weekly_FRA = {
  effect = {
   ww1_FRA_update_crisis = yes
  }
 }
 on_weekly_SOV = {
  effect = {
   ww1_SOV_update_crisis = yes
  }
 }
 on_weekly_AUS = {
  effect = {
   ww1_AUS_update_crisis = yes
  }
 }
 on_weekly_TUR = {
  effect = {
   ww1_TUR_update_crisis = yes
  }
 }
 on_weekly_ITA = {
  effect = {
   ww1_ITA_update_crisis = yes
  }
 }
 on_weekly_USA = {
  effect = {
   ww1_USA_update_crisis = yes
  }
 }
}
